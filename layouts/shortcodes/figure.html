{{- /* Optimized Figure Shortcode with lazy loading Usage: {{< figure src="image.jpg" alt="Description" caption="Caption" >}} */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default "" -}} {{- $caption := .Get "caption" | default "" -}} {{- $class := .Get "class" | default "" -}} {{- $loading := .Get "loading" | default "lazy" -}} {{- $link := .Get "link" | default "" -}} {{- /* Try to get image resource for optimization */ -}} {{- $img := "" -}} {{- with $.Page.Resources.GetMatch $src -}} {{- $img = .Resize "1200x" -}} {{- else -}} {{- $img = resources.Get $src -}} {{- if $img -}} {{- $img = $img.Resize "1200x" -}} {{- end -}} {{- end -}}
