", ")) -}} {{- end -}} {{- else if eq .Section "teaching" -}} {{- /* Course schema for teaching pages */ -}} {{- $schema = dict "@context" "https://schema.org" "@type" "Course" "name" .Title "description" (.Description | default .Summary) "url" .Permalink "provider" (dict "@type" "Organization" "name" "Technical University of Munich" "url" "https://www.tum.de" ) "instructor" (dict "@type" "Person" "name" .Site.Params.author ) -}} {{- else -}} {{- /* Generic webpage schema */ -}} {{- $schema = dict "@context" "https://schema.org" "@type" "WebPage" "name" .Title "description" (.Description | default .Summary | default .Site.Params.description) "url" .Permalink "inLanguage" (.Site.Language.Lang | default "en") -}} {{- end -}}
